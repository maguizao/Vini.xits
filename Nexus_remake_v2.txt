local function AddCorner(instance, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = instance
    -- Garante que o frame recorte os filhos
    if instance:IsA("Frame") or instance:IsA("ScrollingFrame") then
        instance.ClipsDescendants = true
    end
    return corner
end

local function AddStroke(instance, color, thickness, transparency)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness or 1
    stroke.Transparency = transparency or 0
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = instance
    return stroke
end

local function AddShadow(instance, size, color, transparency)
    local shadow = Instance.new("ImageLabel")
    shadow.Name = "Shadow"
    shadow.Size = UDim2.new(1, size * 2, 1, size * 2)
    shadow.Position = UDim2.new(0.5, -size, 0.5, -size)
    shadow.BackgroundTransparency = 1
    shadow.Image = "rbxassetid://1316045115"
    shadow.ImageColor3 = color or Color3.new(0, 0, 0)
    shadow.ImageTransparency = transparency or 0.5
    shadow.ZIndex = -1
    shadow.Parent = instance
    return shadow
end

local function PlayClickSound()

        s.Name = "NexusClickSound"
        s.SoundId = LibraryConfig.ClickSoundId

        s.Parent = NexusUI.ScreenGui or CoreGui
        s:Play()
        s.Ended:Connect(function() s:Destroy() end)
    end)
end

local function MakeDraggable(frame, trigger)
    trigger = trigger or frame
    local dragging, dragInput, dragStart, startPos
    trigger.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    trigger.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            NexusUI.SavedPositions[frame.Name] = {
                frame.Position.X.Scale, frame.Position.X.Offset,
                frame.Position.Y.Scale, frame.Position.Y.Offset
            }
            SaveUIConfig()
        end
    end)
end

function NexusUI.Notify(title, text, duration)

    local order = - (os.clock() * 1000)
    local Frame = Instance.new("Frame")
    Frame.Size = UDim2.new(1, 0, 0, 65)
    Frame.BackgroundColor3 = NexusUI.CurrentTheme.Background
    Frame.BackgroundTransparency = 0.15
    Frame.BorderSizePixel = 0
    Frame.LayoutOrder = order
    Frame.Parent = NexusUI.NotifyList
    Frame.Position = UDim2.new(1, 0, 0, 0)
    Frame.ClipsDescendants = true  -- CORREÇÃO: cantos arredondados
    -- Glass effect
    local glass = Instance.new("Frame")
    glass.Size = UDim2.new(1, 0, 1, 0)
    glass.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    glass.BackgroundTransparency = 0.03
    glass.Parent = Frame

    local Stroke = AddStroke(Frame, NexusUI.CurrentTheme.Accent, 1.5, 0.3)
    AddCorner(Frame, 12)
    AddShadow(Frame, 15, NexusUI.CurrentTheme.ShadowColor, NexusUI.CurrentTheme.ShadowTransparency)

    -- Glow line on top
    local glowLine = Instance.new("Frame")
    glowLine.Size = UDim2.new(0.5, 0, 0, 2)
    glowLine.Position = UDim2.new(0, 0, 0, 0)
    glowLine.BackgroundColor3 = NexusUI.CurrentTheme.Accent
    glowLine.BorderSizePixel = 0
    glowLine.Parent = Frame
    AddCorner(glowLine, 1)
    TweenService:Create(glowLine, TweenInfo.new(2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true), {
        Size = UDim2.new(1, 0, 0, 2)
    }):Play()

    local TitleLbl = Instance.new("TextLabel")
    TitleLbl.Parent = Frame
    TitleLbl.Size = UDim2.new(1, -10, 0, 22)
    TitleLbl.Position = UDim2.new(0, 12, 0, 6)
    TitleLbl.BackgroundTransparency = 1
    TitleLbl.Text = title
    TitleLbl.Font = Enum.Font.GothamBlack
    TitleLbl.TextColor3 = NexusUI.CurrentTheme.Accent
    TitleLbl.TextSize = 14
    TitleLbl.TextXAlignment = Enum.TextXAlignment.Left

    local MsgLbl = Instance.new("TextLabel")
    MsgLbl.Parent = Frame
    MsgLbl.Size = UDim2.new(1, -12, 0, 30)
    MsgLbl.Position = UDim2.new(0, 12, 0, 30)
    MsgLbl.BackgroundTransparency = 1
    MsgLbl.Text = text
    MsgLbl.Font = Enum.Font.GothamSemibold
    MsgLbl.TextColor3 = NexusUI.CurrentTheme.Text
    MsgLbl.TextSize = 12
    MsgLbl.TextXAlignment = Enum.TextXAlignment.Left
    MsgLbl.TextWrapped = true

    Frame.BackgroundTransparency = 1
    TitleLbl.TextTransparency = 1
    MsgLbl.TextTransparency = 1
    Stroke.Transparency = 1

    TweenService:Create(Frame, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 0.15
    }):Play()
    TweenService:Create(TitleLbl, TweenInfo.new(0.4), {TextTransparency = 0}):Play()

    TweenService:Create(Stroke, TweenInfo.new(0.4), {Transparency = 0.3}):Play()

    task.delay(duration or 3.5, function()
        if Frame then
            TweenService:Create(Frame, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Position = UDim2.new(1, 0, 0, 0),
                BackgroundTransparency = 1
            }):Play()
            TweenService:Create(TitleLbl, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
            TweenService:Create(MsgLbl, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
            TweenService:Create(Stroke, TweenInfo.new(0.4), {Transparency = 1}):Play()
            task.wait(0.4)
            Frame:Destroy()
        end
    end)
end

local function ApplyThemeToUI()
    local theme = NexusUI.CurrentTheme
    
    if NexusUI.MainFrame then
        NexusUI.MainFrame.BackgroundColor3 = theme.Background
        NexusUI.MainFrame.BackgroundTransparency = theme.BackgroundTransparency or 0.85
        local stroke = NexusUI.MainFrame:FindFirstChildOfClass("UIStroke")
        if stroke then 
            stroke.Color = theme.Accent
            stroke.Transparency = 0.2
        end
    end
    if NexusUI.TitleLabel then
        NexusUI.TitleLabel.TextColor3 = theme.Accent
    end
    if NexusUI.TabContainer then
        NexusUI.TabContainer.BackgroundColor3 = theme.TabBG
        NexusUI.TabContainer.BackgroundTransparency = theme.TabBGTransparency or 0.05
        local stroke = NexusUI.TabContainer:FindFirstChildOfClass("UIStroke")
        if stroke then
            stroke.Color = theme.BorderColor
            stroke.Transparency = theme.BorderTransparency or 0.12
        end
    end
    -- Abas
    for i, tab in ipairs(NexusUI.Tabs) do
        if tab.Btn then
            tab.Btn.BackgroundColor3 = theme.TabBG
            tab.Btn.BackgroundTransparency = theme.TabBGTransparency or 0.05
            tab.Btn.TextColor3 = theme.TextDim
            if tab.Frame and tab.Frame.Visible then
                tab.Btn.TextColor3 = theme.Text
                tab.Btn.BackgroundColor3 = theme.Accent
                tab.Btn.BackgroundTransparency = 0.15
            end
        end
    end
    -- Underline
    if NexusUI.TabUnderline then
        NexusUI.TabUnderline.BackgroundColor3 = theme.Accent
    end
    if NexusUI.NotifyContainer then
        for _, notify in ipairs(NexusUI.NotifyContainer:GetDescendants()) do
            if notify:IsA("TextLabel") and notify.Text == LibraryConfig.WindowTitle then
                notify.TextColor3 = theme.Accent
            elseif notify:IsA("Frame") and notify:FindFirstChildOfClass("UIStroke") then
                notify.BackgroundColor3 = theme.Background
                notify.BackgroundTransparency = 0.15
                local stroke = notify:FindFirstChildOfClass("UIStroke")
                if stroke then stroke.Color = theme.Accent end
            end
        end
    end
end

local function SaveUIConfig()
    if not LibraryConfig.EnableSaving then return end
    if not writefile then return end
    local config = {
        Theme = LibraryConfig.DefaultTheme,
        Positions = NexusUI.SavedPositions,
        Settings = NexusUI.Config
    }
    local success, encoded = pcall(function()
        return HttpService:JSONEncode(config)
    end)
    if success then
        pcall(function()
            writefile(LibraryConfig.SaveFileName, encoded)
        end)
    end
end

local function LoadUIConfig()
    if not LibraryConfig.EnableSaving then return end
    if not isfile then return end
    if isfile(LibraryConfig.SaveFileName) then
        local content = readfile(LibraryConfig.SaveFileName)
        local success, decoded = pcall(function()
            return HttpService:JSONDecode(content)
        end)
        if success and decoded then
            if decoded.Theme and Themes[decoded.Theme] then
                NexusUI.CurrentTheme = Themes[decoded.Theme]
                LibraryConfig.DefaultTheme = decoded.Theme
            end
            if decoded.Positions then
                NexusUI.SavedPositions = decoded.Positions
            end
            if decoded.Settings then
                NexusUI.Config = decoded.Settings
            end
        end
    end
    if not NexusUI.CurrentTheme then
        NexusUI.CurrentTheme = Themes[LibraryConfig.DefaultTheme] or Themes.Original
    end
end

function NexusUI.SetConfigFileName(fileName)
    if type(fileName) == "string" and fileName ~= "" then
        LibraryConfig.SaveFileName = fileName
        LoadUIConfig()
        for id, element in pairs(NexusUI.RegisteredElements) do
            local savedValue = NexusUI.Config[id]
            if savedValue ~= nil and element.setter then
                element.setter(savedValue)
            end
        end
        ApplyThemeToUI()
        NexusUI.Notify("CONFIG", "Arquivo alterado para: " .. fileName, 3)
        return true
    end
    return false
end

function NexusUI.GetConfigFileName()
    return LibraryConfig.SaveFileName
end

function NexusUI.SetTheme(themeName)
    if Themes[themeName] then
        NexusUI.CurrentTheme = Themes[themeName]
        LibraryConfig.DefaultTheme = themeName
        ApplyThemeToUI()
        SaveUIConfig()
        return true
    end
    return false
end

function NexusUI.SetWindowTitle(title)
    LibraryConfig.WindowTitle = title
    if NexusUI.TitleLabel then
        NexusUI.TitleLabel.Text = title
    end
end

local function RegisterElement(id, type, setter, default)
    NexusUI.RegisteredElements[id] = {type = type, setter = setter, default = default}
end

local function ApplySavedSettings()
    for id, element in pairs(NexusUI.RegisteredElements) do
        local savedValue = NexusUI.Config[id]
        if savedValue ~= nil and element.setter then
            element.setter(savedValue)
        end
    end
end

function NexusUI.SaveAllSettings()
    SaveUIConfig()
    NexusUI.Notify("CONFIG", "Settings saved successfully!", 2)
end

function NexusUI.ResetAllSettings()
    for id, element in pairs(NexusUI.RegisteredElements) do
        if element.default ~= nil and element.setter then
            element.setter(element.default)
        end
    end
    NexusUI.SetTheme("Original")
    SaveUIConfig()
    NexusUI.Notify("CONFIG", "Settings reset to default!", 2)
end

-- ============================================================================
local function ReorderTabs()
    local normalTabs = {}
    local fixedTabs = {}
    for _, tab in ipairs(NexusUI.Tabs) do
        if tab.IsFixed then
            table.insert(fixedTabs, tab)
        else
            table.insert(normalTabs, tab)
        end
    end
    local orderedTabs = {}
    for _, tab in ipairs(normalTabs) do
        table.insert(orderedTabs, tab)
    end
    for _, tab in ipairs(fixedTabs) do
        table.insert(orderedTabs, tab)
    end

    local xOffset = 8
    for i, tab in ipairs(orderedTabs) do
        local btn = tab.Btn
        local width = btn.Size.X.Offset or 90
        btn.Position = UDim2.new(0, xOffset, 0, 4)
        xOffset = xOffset + width + 4
    end

    if NexusUI.TabScroll then
        NexusUI.TabScroll.CanvasSize = UDim2.new(0, xOffset + 20, 0, 0)
    end
    
    UpdateUnderline()
end

local function UpdateUnderline()
    if not NexusUI.TabUnderline then return end
    local activeBtn = nil
    for i, tab in ipairs(NexusUI.Tabs) do
        if tab.Frame and tab.Frame.Visible then
            activeBtn = tab.Btn
            NexusUI.ActiveTabIndex = i
            break
        end
    end
    if activeBtn then
        local btnPos = activeBtn.Position
        local btnSize = activeBtn.Size
        NexusUI.TabUnderline.Position = UDim2.new(btnPos.X.Scale, btnPos.X.Offset + 4, 1, -2)
        NexusUI.TabUnderline.Size = UDim2.new(0, btnSize.X.Offset - 8, 0, 2)
        NexusUI.TabUnderline.BackgroundTransparency = 0
    end
end

function NexusUI.CreateTab(name, isFixed)
    if not NexusUI.MainFrame then error("CreateWindow must be called before CreateTab") end

    local textWidth = math.max(#name * 9 + 30, 80)
    
    local tabBtn = Instance.new("TextButton")
    tabBtn.Size = UDim2.new(0, textWidth, 0, 30)
    tabBtn.Position = UDim2.new(0, 8 + (#NexusUI.Tabs * (textWidth + 4)), 0, 4)
    tabBtn.BackgroundColor3 = NexusUI.CurrentTheme.TabBG
    tabBtn.BackgroundTransparency = NexusUI.CurrentTheme.TabBGTransparency or 0.05
    tabBtn.Text = name
    tabBtn.Font = Enum.Font.GothamSemibold
    tabBtn.TextColor3 = NexusUI.CurrentTheme.TextDim
    tabBtn.TextSize = 13
    tabBtn.AutoButtonColor = false
    tabBtn.Parent = NexusUI.TabScroll
    AddCorner(tabBtn, 8)

    tabBtn.MouseEnter:Connect(function()
        if not tabBtn.Frame or not tabBtn.Frame.Visible then
            TweenService:Create(tabBtn, TweenInfo.new(0.2), {TextColor3 = NexusUI.CurrentTheme.Text}):Play()
            TweenService:Create(tabBtn, TweenInfo.new(0.2), {BackgroundTransparency = 0.1}):Play()
        end
    end)
    tabBtn.MouseLeave:Connect(function()
        if not tabBtn.Frame or not tabBtn.Frame.Visible then
            TweenService:Create(tabBtn, TweenInfo.new(0.2), {TextColor3 = NexusUI.CurrentTheme.TextDim}):Play()
            TweenService:Create(tabBtn, TweenInfo.new(0.2), {BackgroundTransparency = NexusUI.CurrentTheme.TabBGTransparency or 0.05}):Play()
        end
    end)

    local tabFrame = Instance.new("ScrollingFrame")
    tabFrame.Size = UDim2.new(1, 0, 1, 0)
    tabFrame.Position = UDim2.new(0, 0, 0, 0)
    tabFrame.BackgroundTransparency = 1
    tabFrame.ScrollBarThickness = 3
    tabFrame.ScrollBarImageColor3 = NexusUI.CurrentTheme.Accent
    tabFrame.ScrollBarImageTransparency = 0.3
    tabFrame.Visible = false
    tabFrame.AutomaticCanvasSize = Enum.AutomaticSize.Y
    tabFrame.CanvasSize = UDim2.new(0, 0, 0, 0)
    tabFrame.Parent = NexusUI.ContentSlide
    tabFrame.ClipsDescendants = true

    local layout = Instance.new("UIListLayout")
    layout.Padding = UDim.new(0, 8)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Parent = tabFrame

    local padding = Instance.new("UIPadding")
    padding.PaddingTop = UDim.new(0, 6)
    padding.PaddingLeft = UDim.new(0, 6)
    padding.PaddingBottom = UDim.new(0, 6)
    padding.Parent = tabFrame

    local tabEntry = { Btn = tabBtn, Frame = tabFrame, IsFixed = isFixed or false }
    table.insert(NexusUI.Tabs, tabEntry)

    -- Ativa a primeira aba automaticamente se nenhuma estiver visível
    local anyVisible = false
    for _, t in ipairs(NexusUI.Tabs) do
        if t.Frame.Visible then anyVisible = true break end
    end
    if not anyVisible and #NexusUI.Tabs == 1 then
        tabFrame.Visible = true
        tabBtn.TextColor3 = NexusUI.CurrentTheme.Text
        tabBtn.BackgroundColor3 = NexusUI.CurrentTheme.Accent
        tabBtn.BackgroundTransparency = 0.15
        NexusUI.ActiveTabIndex = 1
        task.wait(0.1)
        UpdateUnderline()
    end

    tabBtn.MouseButton1Click:Connect(function()
        PlayClickSound()
        if NexusUI.Animating then return end

        for _, t in ipairs(NexusUI.Tabs) do
            t.Frame.Visible = false
            TweenService:Create(t.Btn, TweenInfo.new(0.3), {
                TextColor3 = NexusUI.CurrentTheme.TextDim,
                BackgroundColor3 = NexusUI.CurrentTheme.TabBG,
                BackgroundTransparency = NexusUI.CurrentTheme.TabBGTransparency or 0.05
            }):Play()
        end

        tabFrame.Visible = true
        TweenService:Create(tabBtn, TweenInfo.new(0.3), {
            TextColor3 = NexusUI.CurrentTheme.Text,
            BackgroundColor3 = NexusUI.CurrentTheme.Accent,
            BackgroundTransparency = 0.15
        }):Play()

        -- Slide animation
        tabFrame.Position = UDim2.new(0.3, 0, 0, 0)
        tabFrame.BackgroundTransparency = 1
        NexusUI.Animating = true
        TweenService:Create(tabFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = UDim2.new(0, 0, 0, 0)
        }):Play()
        
        UpdateUnderline()
        
        task.wait(0.35)
        NexusUI.Animating = false
    end)

    ReorderTabs()
    return tabFrame
end

function NexusUI.CreateToggle(parent, text, default, callback, id)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 36)
    frame.BackgroundColor3 = NexusUI.CurrentTheme.ItemBG
    frame.BackgroundTransparency = NexusUI.CurrentTheme.ItemBGTransparency or 0.06
    frame.Parent = parent
    frame.ClipsDescendants = true
    AddCorner(frame, 10)
    AddStroke(frame, NexusUI.CurrentTheme.BorderColor, 1, NexusUI.CurrentTheme.BorderTransparency or 0.08)
    AddShadow(frame, 8, NexusUI.CurrentTheme.ShadowColor, 0.2)

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 1, 0)
    btn.BackgroundTransparency = 1
    btn.Text = ""
    btn.Parent = frame

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.7, 0, 1, 0)
    label.Position = UDim2.new(0, 14, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = NexusUI.CurrentTheme.Text
    label.Font = Enum.Font.GothamSemibold
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = frame

    -- Switch track
    local track = Instance.new("Frame")
    track.Size = UDim2.new(0, 40, 0, 22)
    track.Position = UDim2.new(1, -50, 0.5, -11)
    track.BackgroundColor3 = Color3.fromRGB(60, 60, 70)
    track.BackgroundTransparency = 0.3
    track.Parent = frame
    AddCorner(track, 11)
    
    -- Switch thumb
    local thumb = Instance.new("Frame")
    thumb.Size = UDim2.new(0, 18, 0, 18)
    thumb.Position = UDim2.new(0, 2, 0.5, -9)
    thumb.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    thumb.Parent = track
    AddCorner(thumb, 9)
    AddShadow(thumb, 6, Color3.new(0,0,0), 0.3)

    local toggled = default
    local setter = function(state)
        toggled = state
        local targetColor = toggled and NexusUI.CurrentTheme.Accent or Color3.fromRGB(60, 60, 70)
        local thumbColor = toggled and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(200, 200, 200)
        local thumbPos = toggled and UDim2.new(0, 20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
        
        TweenService:Create(track, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
            BackgroundColor3 = targetColor,
            BackgroundTransparency = toggled and 0.2 or 0.3
        }):Play()
        TweenService:Create(thumb, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
            Position = thumbPos,
            BackgroundColor3 = thumbColor
        }):Play()
        
        if callback then callback(toggled) end
        if id then
            NexusUI.Config[id] = toggled
            SaveUIConfig()
        end
    end

    btn.MouseButton1Click:Connect(function()
        PlayClickSound()
        setter(not toggled)
    end)

    btn.MouseEnter:Connect(function()
        TweenService:Create(frame, TweenInfo.new(0.2), {
            BackgroundTransparency = (NexusUI.CurrentTheme.ItemBGTransparency or 0.06) - 0.03
        }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(frame, TweenInfo.new(0.2), {
            BackgroundTransparency = NexusUI.CurrentTheme.ItemBGTransparency or 0.06
        }):Play()
    end)

    if id then
        local saved = NexusUI.Config[id]
        if saved ~= nil then
            setter(saved)
        else
            NexusUI.Config[id] = default
        end
        RegisterElement(id, "toggle", setter, default)
    end
    return setter
end

function NexusUI.CreateInput(parent, text, placeholder, defaultVal, callback, id)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 36)
    frame.BackgroundColor3 = NexusUI.CurrentTheme.ItemBG
    frame.BackgroundTransparency = NexusUI.CurrentTheme.ItemBGTransparency or 0.06
    frame.Parent = parent
    frame.ClipsDescendants = true
    AddCorner(frame, 10)
    AddStroke(frame, NexusUI.CurrentTheme.BorderColor, 1, NexusUI.CurrentTheme.BorderTransparency or 0.08)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.4, 0, 1, 0)
    label.Position = UDim2.new(0, 12, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = NexusUI.CurrentTheme.Text
    label.Font = Enum.Font.GothamSemibold
    label.TextSize = 12
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = frame

    local box = Instance.new("TextBox")
    box.Size = UDim2.new(0.5, 0, 0.7, 0)
    box.Position = UDim2.new(0.45, 0, 0.15, 0)
    box.BackgroundColor3 = NexusUI.CurrentTheme.InputBG
    box.BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05
    box.Text = defaultVal or ""
    box.PlaceholderText = placeholder
    box.TextColor3 = NexusUI.CurrentTheme.Accent
    box.Font = Enum.Font.GothamBold
    box.TextSize = 12
    box.Parent = frame
    AddCorner(box, 8)
    local stroke = AddStroke(box, NexusUI.CurrentTheme.BorderColor, 1, 0.1)

    local setter = function(value)
        box.Text = value
        if callback then callback(value) end
        if id then
            NexusUI.Config[id] = value
            SaveUIConfig()
        end
    end

    box.Focused:Connect(function()
        TweenService:Create(stroke, TweenInfo.new(0.2), {
            Color = NexusUI.CurrentTheme.Accent,
            Transparency = 0
        }):Play()
        TweenService:Create(box, TweenInfo.new(0.2), {
            BackgroundTransparency = 0.02
        }):Play()
    end)
    box.FocusLost:Connect(function()
        TweenService:Create(stroke, TweenInfo.new(0.2), {
            Color = NexusUI.CurrentTheme.BorderColor,
            Transparency = 0.1
        }):Play()
        TweenService:Create(box, TweenInfo.new(0.2), {
            BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05
        }):Play()
        if callback then callback(box.Text) end
        if id then
            NexusUI.Config[id] = box.Text
            SaveUIConfig()
        end
    end)

    if id then
        local saved = NexusUI.Config[id]
        if saved ~= nil then
            setter(saved)
        else
            NexusUI.Config[id] = defaultVal
        end
        RegisterElement(id, "input", setter, defaultVal)
    end
    return box, setter
end

    keyBtn.Position = UDim2.new(0.6, 0, 0.15, 0)
    keyBtn.BackgroundColor3 = NexusUI.CurrentTheme.InputBG
    keyBtn.BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05
    keyBtn.Text = defaultKey
    keyBtn.TextColor3 = NexusUI.CurrentTheme.Accent
    keyBtn.Font = Enum.Font.GothamBold
    keyBtn.TextSize = 12
    keyBtn.Parent = frame
    AddCorner(keyBtn, 8)
    local stroke = AddStroke(keyBtn, NexusUI.CurrentTheme.BorderColor, 1, 0.1)

    local setter = function(key)
        keyBtn.Text = key
        if callback then callback(key) end
        if id then
            NexusUI.Config[id] = key
            SaveUIConfig()
        end
    end

    local listening = false
    local connection

    keyBtn.MouseButton1Click:Connect(function()
        PlayClickSound()
        listening = true
        keyBtn.Text = "..."
        keyBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
        stroke.Color = Color3.fromRGB(255, 50, 50)
        stroke.Transparency = 0

        if connection then connection:Disconnect() end
        connection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if listening and not gameProcessed then
                local keyName = tostring(input.KeyCode):gsub("Enum.KeyCode.", "")
                setter(keyName)
                listening = false
                connection:Disconnect()
            end
        end)

        task.delay(10, function()
            if listening then
                listening = false
                keyBtn.Text = defaultKey
                keyBtn.TextColor3 = NexusUI.CurrentTheme.Accent
                stroke.Color = NexusUI.CurrentTheme.BorderColor
                stroke.Transparency = 0.1
                if connection then connection:Disconnect() end
            end
        end)
    end)

    if id then
        local saved = NexusUI.Config[id]
        if saved ~= nil then
            setter(saved)
        else
            NexusUI.Config[id] = defaultKey
        end
        RegisterElement(id, "keybind", setter, defaultKey)
    end
    return keyBtn, setter
end

-- ============================================================================
-- DROPDOWN PREMIUM (com animação e glassmorphism)
-- ============================================================================
function NexusUI.CreateDropdown(parent, text, options, default, callback, id)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 36)
    frame.BackgroundColor3 = NexusUI.CurrentTheme.ItemBG
    frame.BackgroundTransparency = NexusUI.CurrentTheme.ItemBGTransparency or 0.06
    frame.Parent = parent
    frame.ClipsDescendants = true
    AddCorner(frame, 10)
    AddStroke(frame, NexusUI.CurrentTheme.BorderColor, 1, NexusUI.CurrentTheme.BorderTransparency or 0.08)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.4, 0, 1, 0)
    label.Position = UDim2.new(0, 12, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = NexusUI.CurrentTheme.Text
    label.Font = Enum.Font.GothamSemibold
    label.TextSize = 12
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = frame

    local dropdownBtn = Instance.new("TextButton")
    dropdownBtn.Size = UDim2.new(0.5, 0, 0.7, 0)
    dropdownBtn.Position = UDim2.new(0.45, 0, 0.15, 0)
    dropdownBtn.BackgroundColor3 = NexusUI.CurrentTheme.InputBG
    dropdownBtn.BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05
    dropdownBtn.Text = default or options[1] or ""
    dropdownBtn.TextColor3 = NexusUI.CurrentTheme.Accent
    dropdownBtn.Font = Enum.Font.GothamBold
    dropdownBtn.TextSize = 12
    dropdownBtn.Parent = frame
    AddCorner(dropdownBtn, 8)
    local stroke = AddStroke(dropdownBtn, NexusUI.CurrentTheme.BorderColor, 1, 0.1)

    -- Seta do dropdown
    local arrow = Instance.new("TextLabel")
    arrow.Size = UDim2.new(0, 20, 1, 0)
    arrow.Position = UDim2.new(1, -24, 0, 0)
    arrow.BackgroundTransparency = 1
    arrow.Text = "▾"
    arrow.TextColor3 = NexusUI.CurrentTheme.TextDim
    arrow.Font = Enum.Font.GothamBold
    arrow.TextSize = 14
    arrow.TextXAlignment = Enum.TextXAlignment.Center
    arrow.Parent = dropdownBtn

    local dropdownList = nil
    local isOpen = false

    local function closeDropdown()
        if dropdownList then
            TweenService:Create(dropdownList, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
                BackgroundTransparency = 1,
                Size = UDim2.new(0, 0, 0, 0)
            }):Play()
            TweenService:Create(arrow, TweenInfo.new(0.2), {Text = "▾"}):Play()
            task.wait(0.2)
            dropdownList:Destroy()
            dropdownList = nil
        end
        isOpen = false
    end

    local function openDropdown()
        if isOpen then return end
        isOpen = true
        TweenService:Create(arrow, TweenInfo.new(0.2), {Text = "▴"}):Play()

        local dropdownContainer = Instance.new("Frame")
        dropdownContainer.Name = "DropdownContainer"
        dropdownContainer.Size = UDim2.new(0, 0, 0, 0)
        dropdownContainer.BackgroundColor3 = NexusUI.CurrentTheme.Background
        dropdownContainer.BackgroundTransparency = 0.15
        dropdownContainer.BorderSizePixel = 0
        dropdownContainer.ZIndex = 999
        dropdownContainer.Parent = NexusUI.ScreenGui
        dropdownContainer.ClipsDescendants = true
        AddCorner(dropdownContainer, 12)
        AddStroke(dropdownContainer, NexusUI.CurrentTheme.Accent, 1.5, 0.3)
        AddShadow(dropdownContainer, 15, NexusUI.CurrentTheme.ShadowColor, NexusUI.CurrentTheme.ShadowTransparency)

        -- Glass effect
        local glass = Instance.new("Frame")
        glass.Size = UDim2.new(1, 0, 1, 0)
        glass.BackgroundColor3 = Color3.fromRGB(255,255,255)
        glass.BackgroundTransparency = 0.02
        glass.Parent = dropdownContainer

        local maxHeight = math.min(#options * 32, 160)
        dropdownContainer.Size = UDim2.new(0, 240, 0, maxHeight)
        TweenService:Create(dropdownContainer, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 240, 0, maxHeight)
        }):Play()

        local btnAbsPos = dropdownBtn.AbsolutePosition
        local parentAbsPos = frame.AbsolutePosition
        local xOffset = btnAbsPos.X - parentAbsPos.X
        dropdownContainer.Position = UDim2.new(0, xOffset, 0, 38)

        local scroll = Instance.new("ScrollingFrame")
        scroll.Size = UDim2.new(1, -4, 1, -4)
        scroll.Position = UDim2.new(0, 2, 0, 2)
        scroll.BackgroundTransparency = 1
        scroll.ScrollBarThickness = 3
        scroll.ScrollBarImageColor3 = NexusUI.CurrentTheme.Accent
        scroll.ScrollBarImageTransparency = 0.3
        scroll.CanvasSize = UDim2.new(0, 0, 0, #options * 32)
        scroll.AutomaticCanvasSize = Enum.AutomaticSize.Y
        scroll.Parent = dropdownContainer
        scroll.ClipsDescendants = true

        local layout = Instance.new("UIListLayout")
        layout.Padding = UDim.new(0, 2)
        layout.SortOrder = Enum.SortOrder.LayoutOrder
        layout.Parent = scroll

        for _, opt in ipairs(options) do
            local optBtn = Instance.new("TextButton")
            optBtn.Size = UDim2.new(1, 0, 0, 30)
            optBtn.BackgroundColor3 = NexusUI.CurrentTheme.InputBG
            optBtn.BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05
            optBtn.Text = opt
            optBtn.TextColor3 = NexusUI.CurrentTheme.Text
            optBtn.Font = Enum.Font.GothamSemibold
            optBtn.TextSize = 12
            optBtn.Parent = scroll
            optBtn.ClipsDescendants = true
            AddCorner(optBtn, 6)
            AddStroke(optBtn, NexusUI.CurrentTheme.BorderColor, 1, 0.05)

            optBtn.MouseEnter:Connect(function()
                TweenService:Create(optBtn, TweenInfo.new(0.15), {
                    BackgroundTransparency = 0.02,
                    BackgroundColor3 = NexusUI.CurrentTheme.Accent
                }):Play()
                TweenService:Create(optBtn, TweenInfo.new(0.15), {
                    TextColor3 = Color3.fromRGB(255, 255, 255)
                }):Play()
            end)
            optBtn.MouseLeave:Connect(function()
                TweenService:Create(optBtn, TweenInfo.new(0.15), {
                    BackgroundTransparency = NexusUI.CurrentTheme.InputBGTransparency or 0.05,
                    BackgroundColor3 = NexusUI.CurrentTheme.InputBG
                }):Play()
                TweenService:Create(optBtn, TweenInfo.new(0.15), {
                    TextColor3 = NexusUI.CurrentTheme.Text
                }):Play()
            end)

            optBtn.MouseButton1Click:Connect(function()
                PlayClickSound()
                dropdownBtn.Text = opt
                if callback then callback(opt) end
                if id then
                    NexusUI.Config[id] = opt
                    SaveUIConfig()
                end
                closeDropdown()
            end)
        end

        local function onGlobalClick(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                local mousePos = input.Position
                local btnAbsPos = dropdownBtn.AbsolutePosition
                local btnSize = dropdownBtn.AbsoluteSize
                local contAbsPos = dropdownContainer.AbsolutePosition
                local contSize = dropdownContainer.AbsoluteSize

                local insideBtn = (mousePos.X >= btnAbsPos.X and mousePos.X <= btnAbsPos.X + btnSize.X and
                                   mousePos.Y >= btnAbsPos.Y and mousePos.Y <= btnAbsPos.Y + btnSize.Y)
                local insideContainer = (mousePos.X >= contAbsPos.X and mousePos.X <= contAbsPos.X + contSize.X and
                                         mousePos.Y >= contAbsPos.Y and mousePos.Y <= contAbsPos.Y + contSize.Y)

                if not insideBtn and not insideContainer then
                    closeDropdown()
                end
            end
        end

        local connection
        connection = UserInputService.InputBegan:Connect(function(input)
            onGlobalClick(input)
            if not isOpen then
                if connection then connection:Disconnect() end
            end
        end)

        dropdownContainer.Destroying:Connect(function()
            if connection then connection:Disconnect() end
        end)

        dropdownList = dropdownContainer
    end

    dropdownBtn.MouseButton1Click:Connect(function()
        PlayClickSound()
        if isOpen then
            closeDropdown()
        else
            openDropdown()
        end
    end)

    local setter = function(value)
        if table.find(options, value) then
            dropdownBtn.Text = value
            if callback then callback(value) end
            if id then
                NexusUI.Config[id] = value
                SaveUIConfig()
            end
        end
    end

    if id then
        local saved = NexusUI.Config[id]
        if saved ~= nil and table.find(options, saved) then
            setter(saved)
        else
            NexusUI.Config[id] = default or options[1]
            setter(default or options[1])
        end
        RegisterElement(id, "dropdown", setter, default or options[1])
    end
    return setter
end

    if not NexusUI.CreditsContainer then
        NexusUI.Notify("CREDITS", "Aba de créditos não foi criada.", 3)
        return
    end
    local card = Instance.new("Frame")
    card.Size = UDim2.new(1, -10, 0, 95)

    card.BackgroundTransparency = NexusUI.CurrentTheme.ItemBGTransparency or 0.06
    card.Parent = NexusUI.CreditsContainer
    card.ClipsDescendants = true

    AddStroke(card, NexusUI.CurrentTheme.BorderColor, 1, NexusUI.CurrentTheme.BorderTransparency or 0.08)
    AddShadow(card, 10, NexusUI.CurrentTheme.ShadowColor, 0.2)

    if imageId and imageId ~= "" then

        img.Size = UDim2.new(0, 60, 0, 60)

    local nameLabel = Instance.new("TextLabel")
    nameLabel.Size = UDim2.new(0.6, 0, 0, 26)
    nameLabel.Position = UDim2.new(0, 84, 0, 10)
    nameLabel.BackgroundTransparency = 1

function NexusUI.ClearCredits()
    if NexusUI.CreditsContainer then
        for _, child in ipairs(NexusUI.CreditsContainer:GetChildren()) do
            if child:IsA("Frame") then
                child:Destroy()
            end
        end
    end
end

-- ============================================================================
-- DESTRUIÇÃO
-- ============================================================================
function NexusUI.Destroy()
    if NexusUI.ScreenGui then
        NexusUI.ScreenGui:Destroy()
    end
    if NexusUI.NotifyContainer then
        NexusUI.NotifyContainer:Destroy()
    end
    NexusUI.Tabs = {}
    NexusUI.SavedPositions = {}
end

    task.wait(0.1)
    ApplySavedSettings()
end)

return NexusUI
